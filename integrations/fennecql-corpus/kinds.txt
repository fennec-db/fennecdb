# FennecQL's texts by kind, written by hand beside the frozen corpora
# (docs/design/fennecql.md#the-corpus): at least five of every kind of
# statement fennec-ql reads, of every SQL spelling of one and of every SQL
# word it refuses naming what to write (tools/fennecql_corpus.py's
# REQUIRED_KINDS) -- the kinds the repository's own texts hold few of.
#
# A text a line; a line beginning with `#` is a comment. The corpus step
# against main reads every line with both trees; `--freeze <tag>` reads
# each with the tag's own parser and keeps it whatever it read, a refusal
# too, so a text a later release reads that the tag refused is told and
# free. A line added reaches a frozen corpus at the next tag's freeze.

# create statement
create statement recent() as get notes select id, title order at desc limit 10
create statement by_tag($tag text) as get notes where tag = $tag select id, title
create statement years($from int, $n int = 2) comment 'oldest first' as get docs select title, year where year >= $from order year limit $n
create statement mine() claims (sub text) as get accounts select id where owner = $jwt.sub
create statement purge() public as del notes where done = true
create statement tidy every 1d as del orders where ref = 'void'
create statement nightly at '30 2 * * *' in 'Europe/Berlin' scans as del holds where state = 'lapsed'
create statement took($k text from event.id) on hook stripe as insert taken {k: $k}
create statement pay($from int, $to int, $n int) public as begin set accounts {balance: balance - $n} where id = $from require 1; set accounts {balance: balance + $n} where id = $to require 1; end
